/* ================= V9: forge and item tiers, enemy cut-ins, who-is-acting colours ================= */
var TIER_N=['Normal','Refinada','Màgica','Èpica','Superèpica'];
var TIER_M=[1,1.6,2.4,3.4,4.8];
var TIER_C=['#7D8597','#2E8BE6','#9B59E0','#E0572B','#C98A00'];
var FORGE_COST=[30,60,120,240];
function tierOf(id){return Math.min(4,+(String(id).split('#')[1]||0))}
function baseOf(id){return String(id).split('#')[0]}
function IT(id){if(id==null)return null;var b=ITEMS[baseOf(id)];if(!b)return null;var t=tierOf(id),m=TIER_M[t];
  return {n:b.n+(t?' '+'★'.repeat(t):''),slot:b.slot,atk:b.atk*m,def:b.def*m,hp:b.hp*m,rar:(b.relic?'Tresor · ':'')+TIER_N[t],relic:b.relic,tier:t,base:baseOf(id)}}
function itemStats(it){return [it.atk?'Atac +'+Math.round(it.atk*3)+'%':'',it.def?'Defensa +'+Math.round(it.def*3)+'%':'',it.hp?'Vida +'+Math.round(it.hp*2)+'%':''].filter(Boolean).join(' · ')}
function itemGroups(){var g={},order=[];G.items.forEach(function(id){if(!IT(id))return;if(!g[id]){g[id]=0;order.push(id)}g[id]++});
  return order.sort(function(a,b){var A=IT(a),B=IT(b);return (A.slot>B.slot?1:A.slot<B.slot?-1:0)||(A.base>B.base?1:A.base<B.base?-1:0)||tierOf(b)-tierOf(a)}).map(function(id){return {id:id,n:g[id]}})}
function forgeOnce(id){var t=tierOf(id);if(t>=4)return false;var cnt=G.items.filter(function(x){return x===id}).length;if(cnt<4||G.coins<FORGE_COST[t])return false;
  for(var k=0;k<4;k++)G.items.splice(G.items.indexOf(id),1);G.coins-=FORGE_COST[t];G.items.push(baseOf(id)+'#'+(t+1));return true}
function forgeAll(){var n=0,again=true,guard=0;while(again&&guard++<200){again=false;var gs=itemGroups();for(var i=0;i<gs.length;i++){if(gs[i].n>=4&&forgeOnce(gs[i].id)){n++;again=true;break}}}return n}
function scMotxilla(){
  var q=bonusPct(G.equip);
  var slots=SLOTS.map(function(x){var id=G.equip[x[0]],it=IT(id);return '<div class="box" style="min-height:96px;'+(it?'border:3px solid '+TIER_C[it.tier]:'border:2px dashed #B9B3C6')+'"><span class="muted small">'+x[1]+'</span>'+(it?'<b style="font-size:13px;text-align:center">'+esc(it.n)+'</b><span class="muted small" style="text-align:center">'+esc(itemStats(it))+'</span><button type="button" class="btn sm light" style="min-height:36px" data-act="unequip" data-slot="'+x[0]+'">Treu</button>':'<span class="muted small">buit</span>')+'</div>'}).join('');
  var groups=itemGroups(),ready=groups.filter(function(g){return g.n>=4&&tierOf(g.id)<4});
  var forge='<div class="flat" style="gap:8px;background:#FFF1DA"><h2>Forja</h2><p class="small">Amb <b>4 objectes iguals</b> en fas <b>1 de la categoria següent</b>: Normal → Refinada → Màgica → Èpica → Superèpica. Cada categoria dona molt més (Refinada ×1,6 · Màgica ×2,4 · Èpica ×3,4 · Superèpica ×4,8). Costa '+FORGE_COST.join(', ')+' monedes segons la categoria.</p>'+
    (ready.length?'<button type="button" class="btn gold" data-act="forgeAll">Forja-ho tot ('+ready.length+' grups a punt)</button>':'<p class="muted small">Ara no tens 4 objectes iguals de cap mena.</p>')+'</div>';
  var inv=groups.map(function(g){var it=IT(g.id),t=it.tier,sl=it.slot==='anell'?['anell1','anell2']:[it.slot],cost=t<4?FORGE_COST[t]:0;
    return '<div class="flat" style="background:var(--soft);gap:6px;outline:3px solid '+TIER_C[t]+'"><div class="row between"><b>'+esc(it.n)+(g.n>1?' ×'+g.n:'')+'</b><span class="tag" style="background:'+TIER_C[t]+';color:#fff">'+esc(it.rar)+'</span></div><span class="muted small">'+esc(itemStats(it))+'</span>'+
      sl.map(function(s){var eqNew=Object.assign({},G.equip);eqNew[s]=g.id;var cur=G.equip[s];
        return '<div class="row" style="gap:6px"><button type="button" class="btn sm" data-act="equip" data-id="'+esc(g.id)+'" data-slot="'+s+'">'+(cur?'Canvia per aquest':'Equipa')+(sl.length>1?' ('+(s==='anell1'?'Anell 1':'Anell 2')+')':'')+'</button></div><div class="row" style="gap:4px">'+deltaChips(eqNew)+'</div>'}).join('')+
      (t<4?'<div class="row" style="gap:6px"><button type="button" class="btn sm gold" data-act="forge" data-id="'+esc(g.id)+'"'+(g.n>=4&&G.coins>=cost?'':' disabled')+'>Forja 4 → 1 '+esc(TIER_N[t+1])+' ('+cost+' monedes)</button><span class="muted small">'+(g.n>=4?(G.coins>=cost?'':'Et falten monedes'):'En tens '+g.n+' de 4')+'</span></div>':'<span class="muted small">Ja és Superèpica: la millor categoria.</span>')+'</div>'}).join('');
  var food=Object.keys(ELS).map(function(k){return '<div class="box">'+elIcon(k,26,ELS[k].c)+'<span class="small" style="text-align:center">'+esc(FOOD[k])+'</span><b>'+(G.food[k]||0)+'</b></div>'}).join('');
  return '<h1>Motxilla</h1>'+tip('motxilla','Els objectes que portes fan més forta cada bestiola de l’equip. Abans d’equipar, mira les fletxes: ▲ verd vol dir que millora, ▼ vermell que empitjora. Amb 4 objectes iguals, a la Forja en fas un de millor.')+
    '<div class="flat" style="align-items:center">'+hero(48)+'<div class="row" style="justify-content:center"><span class="tag">Atac +'+Math.round(q.atk)+'%</span><span class="tag">Defensa +'+Math.round(q.def)+'%</span><span class="tag">Vida +'+Math.round(q.hp)+'%</span></div></div>'+
    bestBlock()+'<div class="grid4">'+slots+'</div>'+forge+'<h2>A la motxilla</h2>'+(inv||'<p class="muted">Encara no tens objectes. En trobaràs explorant.</p>')+'<h2>Menjar</h2><div class="grid3">'+food+'</div>';
}
/* who is acting: green for your team, red for the enemy; big enemy moves get their own cut-in */
var US_EV=['hit','heal','regen','special','combo','dot','stunned','foeDown'],THEM_EV=['ehit','aoe','warn','enrage','wave','teamDown'];
function showEvent(e){
  var cap=$('caption');if(cap){cap.classList.toggle('them',THEM_EV.indexOf(e.t)>=0);cap.classList.toggle('us',US_EV.indexOf(e.t)>=0)}
  var big=(e.t==='ehit'&&e.strong)||e.t==='aoe'||e.t==='enrage';
  if(big){enemyCut(e);setTimeout(function(){showEventNow(e)},RM?100:950);return}
  showEventNow(e);
  if(e.t==='ehit'){var fa=$('foe-art');floatNum(fa,'ATACA!','tagtxt')}
}
function enemyCut(e){
  var r=run(),b=r&&r.event,st=$('stage');if(!st||!b||!b.enemy)return;var en=b.enemy;
  var ci=document.createElement('div');ci.className='cutin enemy';ci.style.setProperty('--cc',ELS[en.el].c);
  ci.innerHTML='<div class="lines"></div><div class="side">ENEMIC</div><div class="who foeIn">'+(en.wild?spArt(en.wild,'',118):monster(en.el,118,en.guardian))+'</div><div class="sname">'+(e.t==='aoe'?'ATAC A TOTHOM!':e.t==='enrage'?'S’ENFADA!':'COP FORT!')+'</div>';
  st.appendChild(ci);anim(st,'shake');setTimeout(function(){ci.remove();anim($('squad'),'shake')},RM?100:950);
}
